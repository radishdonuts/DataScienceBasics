== SPARK SETUP ==
(We now use 4.0.1)
wget https://downloads.apache.org/spark/spark-3.5.6/spark-3.5.6-bin-hadoop3.tgz
https://dlcdn.apache.org/spark/spark-4.0.1/spark-4.0.1-bin-hadoop3.tgz

tar -xzvf spark-3.5.6-bin-hadoop3.tgz
sudo mv spark-3.5.6-bin-hadoop3 /opt/spark

nano ~/.bashrc
export SPARK_HOME=/opt/spark
export PATH=$PATH:$SPARK_HOME/bin:$SPARK_HOME/sbin
source ~/.bashrc

sudo useradd spark
cat /etc/passwd

sudo chown -R spark:spark /opt/spark
ls -la /opt/

sudo nano /etc/systemd/system/spark-master.service

[Unit]
Description=Apache Spark Master
After=network.target

[Service]
Type=forking
User=spark
Group=spark
ExecStart=/opt/spark/sbin/start-master.sh
ExecStop=/opt/spark/sbin/stop-master.sh

[Install]
WantedBy=multi-user.target

sudo nano /etc/systemd/system/spark-slave.service

[Unit]
Description=Apache Spark Slave
After=network.target

[Service]
Type=forking
User=spark
Group=spark
ExecStart=/opt/spark/sbin/start-slave.sh spark://your-server-ip:7077
ExecStop=/opt/spark/sbin/stop-slave.sh

[Install]
WantedBy=multi-user.target

sudo systemctl daemon-reload

tmux

sudo systemctl start spark-master
sudo systemctl enable spark-master
sudo systemctl status spark-master

sudo systemctl start spark-worker
sudo systemctl enable spark-worker
sudo systemctl status spark-worker
spark-shell
